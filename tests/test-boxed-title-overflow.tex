  boxrule=0pt,
  arc=2pt,

  borderline west={3pt}{0pt}{keypointframe},
  coltitle=black,
  attach boxed title to top left={yshift=-2mm, xshift=4mm},
  minipage boxed title*=-14mm,
  boxed title style={
    colback=keypointbg,
    colframe=keypointframe,
    boxrule=0.5pt,
    arc=1pt
  }
}
